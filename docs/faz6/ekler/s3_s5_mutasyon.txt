run() { # run <ad> <açıklama> <pytest hedefleri...>
  local name=$1 desc=$2; shift 2
  local log=$W/$name.log
  "$PY" -m pytest -q -p no:cacheprovider --no-header -rfE "$@" 2>&1 | sed -E "$MASK" > "$log"; local rc=${PIPESTATUS[0]}
  echo "== $name  rc=$rc  :: $desc" >> "$OUT"
  grep -E '^(FAILED|ERROR) ' "$log" | sed -E 's/^(FAILED|ERROR) tests\/test_faz6_oot.py::/  \1 /' | cut -c1-200 >> "$OUT"
  grep -E 'passed|failed' "$log" | tail -1 | sed 's/^/  /' >> "$OUT"
}
mutate() { # mutate <dosya> <ad>; stdin: python (s -> s)
  cp "$C/$1" "$W/backup"

}
