OUT=$W/out.txt; : > "$OUT"

run() { # run <ad> <açıklama> <pytest hedefleri...>
  local name=$1 desc=$2; shift 2
  local log=$W/$name.log
  "$PY" -m pytest -q -p no:cacheprovider --no-header -rfE "$@" 2>&1 | sed -E "$MASK" > "$log"; local rc=${PIPESTATUS[0]}
  echo "== $name  rc=$rc  :: $desc" >> "$OUT"
  grep -E '^(FAILED|ERROR) ' "$log" | sed -E 's/^(FAILED|ERROR) tests\/test_faz6_oot.py::/  \1 /' | cut -c1-230 >> "$OUT"
  grep -E '^[0-9]+ (passed|failed)|^[0-9]+ failed|passed in|failed in' "$log" | tail -1 | sed 's/^/  /' >> "$OUT"
}
mutate() { # mutate <dosya> <ad>; stdin: python (s -> s)
  cp "$C/$1" "$W/backup"
  "$PY" - "$C/$1" "$2" || { echo "MUTASYON UYGULANAMADI: $2" >> "$OUT"; exit 1; }
}
restore() { cp "$W/backup" "$C/$1"; }
